def native(value):
    kind=type(value).__name__
    if value is None or type(value) in (bool,int,str):return {'type':kind,'value':value}
    if type(value) is float:return {'type':'float','hex':value.hex()}
    if type(value) in (list,tuple):return {'type':kind,'items':[native(v) for v in value]}
    if type(value) is dict:return {'type':'dict','items':[[native(k),native(v)] for k,v in value.items()]}
    raise TypeError(('unsupported native value',kind))

def delta(before,after):return {key:after.get(key,0)-before.get(key,0) for key in sorted(set(before)|set(after))}

def profile(frame,event,value):
    path=frame.f_code.co_filename.replace('\\','/').lower();name=frame.f_code.co_name
    if event=='call' and '/rouge/' in path:
        relative='rouge/'+path.split('/rouge/',1)[1]
        qualified=relative+':'+name;library_entries[qualified]=library_entries.get(qualified,0)+1
        key=None
        if path.endswith('/rouge/app.py') and name=='calculate':key='MainWindow.calculate'
        elif path.endswith('/rouge/damage.py') and name=='calculate_damage':key='calculate_damage'
        elif path.endswith('/rouge/estimate.py') and name=='format_estimate':key='format_estimate'
        elif path.endswith('/rouge/reporting.py') and name=='format_report':key='format_report_technical' if frame.f_locals['technical'] else 'format_report_default'
        elif path.endswith('/rouge/run_state.py') and name in ('__init__','apply','load'):key='RunState.'+name
        if key:entry_counts[key]=entry_counts.get(key,0)+1
        if key=='calculate_damage':
            record={'sequence':len(API_events)+1,'step':current_step,
                'input':copy.deepcopy(frame.f_locals['scenario']),
                'input_native_before':native(frame.f_locals['scenario']),'outcome':'pending'}
            API_events.append(record);pending_API[id(frame)]=record
    elif event=='return' and id(frame) in pending_API:
        record=pending_API.pop(id(frame));record['input_native_after']=native(frame.f_locals['scenario'])
        record['caller_native_unchanged']=record['input_native_after']==record['input_native_before']
        record['outcome']='returned_dict' if type(value) is dict else 'returned_None_or_exception_unwind'
    return profile

        'passed':receipt['passed'],'current_step':current_step}
    decoded=json.dumps(payload,ensure_ascii=False,allow_nan=False).encode('utf-8')
    compressed=gzip.compress(decoded,mtime=0);CHECKPOINT.write_bytes(compressed)
    receipt['lossless_records']={'file':CHECKPOINT.name,'records':len(states),'bytes':len(compressed),
        'sha256':hashlib.sha256(compressed).hexdigest(),'decoded_bytes':len(decoded),
        'decoded_sha256':hashlib.sha256(decoded).hexdigest()}
